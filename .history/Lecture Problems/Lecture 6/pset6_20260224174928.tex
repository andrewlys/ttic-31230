\section*{Problem 1: Moving Averages}
Consider $x_0, x_1, \ldots$ and two running averages defined as, for $0 < \beta < 1$ and $\gamma > 0$. 
\begin{align*}
    y_0 &= 0\\
    y_{t + 1} &= \beta y_t + (1 - \beta) x_t\\
    \\
    z_0 &= 0 \\
    z_{t + 1} &= \beta z_t + \gamma x_t
\end{align*}
\subsection*{Part (a)} Suppose $x_t$ i.i.d. from distribution with mean vector $\bar{x}$
Give values for 
\[
\bar{y} = \lim_{t \to \infty} \ee{y_t}
\]
and 
\[
\bar{z} = \lim_{t\to \infty} \ee{z_t}
\]
as functions of $\beta, \gamma,$ and $\bar{x}$. 

\begin{proof}
    \begin{align*}
        \ee{y_2} &= \beta \ee{y_1} + (1 - \beta) \bar{x} \\
        &= \beta( \beta \ee{y_0} + (1 - \beta) \bar{x}) + (1 - \beta) \bar{x} \\
        &= \beta^2 \ee{y_0} + \beta(1 - \beta) \bar{x} + (1 - \beta) \bar{x} \\
        &= (1 - \beta) \bar{x} \sum_{i = 0}^1 \beta^i\\
        \implies \ee{y_t} &= (1 - \beta) \bar{x} \sum_{i = 0}^{t - 1} \beta^i\\
        \implies \bar{y} = \lim_{t \to \infty} \ee{y_t} &= (1 - \beta) \bar{x} \sum_{i = 0}^{\infty} \beta^i\\
        &= \bar{x}
    \end{align*}
    For $z_t$, we have
    \begin{align*}
        \ee{z_3} &= \beta \ee{z_2} + \gamma \bar{x} \\
        &= \beta( \beta \ee{z_1} + \gamma \bar{x}) + \gamma \bar{x} \\
        &= \beta^2 \ee{z_1} + \beta \gamma \bar{x} + \gamma \bar{x} \\
        &= \beta^2 ( \beta \ee{z_0} + \gamma \bar{x}) + \beta \gamma \bar{x} + \gamma \bar{x} \\
        &= \beta^3 \ee{z_0} + \gamma \bar{x} \sum_{i = 0}^2 \beta^i\\
        \implies \ee{z_t} &=  \gamma \bar{x} \sum_{i = 0}^{t - 1} \beta^i\\
        \implies \bar{z} = \lim_{t \to \infty} \ee{z_t} &= \gamma \bar{x} \sum_{i = 0}^{\infty} \beta^i\\
        &= \frac{\gamma}{1 - \beta} \bar{x}
    \end{align*}
\end{proof}

\subsection*{Part (b)} Express $z_t$ as a function of $y_t, \beta$, and $\gamma$.
\begin{proof}
    From part (a) we have:
    \begin{align*}
        z_t &= \gamma \sum_{i = 0} ^{t - 1} \beta^i x_{t - 1 - i} \\
        y_t &= (1 - \beta)\sum_{i = 0} ^{t - 1} \beta^i x_{t - 1 - i}
    \end{align*}
    This implies that 
    \[
    z_t = \frac{\gamma}{1-\beta} y_t
    \]
\end{proof}


\section*{Problem 2: Variance of an exponential moving average}
Consider a running average for computing $\hat{\mu}_1, \ldots, \hat{\mu_t}$ from $x_1, \ldots, x_t$
\begin{align*}
    \hat{\mu}_0 &= 0\\
    \hat{\mu}_t &= \left(
    1 - \frac1{N} 
    \right) \hat{\mu}_{t - 1} + \frac1{N} x_t
\end{align*}
\subsection*{Part (a)} Suppose $x_t$ are i.i.d. with variance $\sigma_x^2$. 
Find 
\[
\sigma_{\hat{\mu}, \infty}^2 = \lim_{t\to\infty} \Var{\hat{\mu}_t}
\]
\begin{proof}
    Since $x_t \ind \hat{\mu}_{t - 1}$, we have
    \begin{align*}
        \Var{\hat{\mu}_t} &= \left(
            1 - \frac1{N}
        \right)^2\Var{ (\hat{\mu}_{t - 1})} + \frac1{N^2} \Var{x_t} \\
        &= \left(1 - \frac1{N}\right)^2 \Var{(\hat{\mu}_{t - 1})} + \frac{\sigma_x^2}{N^2}
    \end{align*}
    Taking limits on both sides, we have
    \begin{align*}
        \sigma_{\hat{\mu}, \infty}^2 &= \left(1 - \frac1{N}\right)^2 \sigma_{\hat{\mu}, \infty}^2 + \frac{\sigma_x^2}{N^2}\\ 
        \left(
            1 - \left(1 - \frac1{N}\right)^2
        \right) \sigma_{\hat{\mu}, \infty}^2 &= \frac{\sigma_x^2}{N^2} \\
        \left(
            1 - 1 + \frac1{N}
        \right)\left(
            1 + 1 - \frac1{N}
        \right) \sigma_{\hat{\mu}, \infty}^2 &= \frac{\sigma_x^2}{N^2} \\
        \frac{2N - 1}{N^2} \sigma_{\hat{\mu}, \infty}^2 &= \frac{\sigma_x^2}{N^2} \\
        \sigma_{\hat{\mu}, \infty}^2 &= \frac{\sigma_x^2}{2N - 1}
    \end{align*}
\end{proof}
